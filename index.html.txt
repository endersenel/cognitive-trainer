    <script>
        // Web Audio API Kurulumu (Düşük gecikmeli ses üretimi)
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx = null;

        function playSound(freq, type = 'sine', duration = 0.15) {
            if (!audioCtx) audioCtx = new AudioContext();
            if (audioCtx.state === 'suspended') audioCtx.resume();
            
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = type;
            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration);
            
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + duration);
        }

        function playGoSound() { playSound(880, 'sine', 0.15); }      // Yüksek Do sesi (GO)
        function playNoGoSound() { playSound(150, 'sawtooth', 0.3); } // Düşük Cızz sesi (NO-GO)

        // Test Değişkenleri
        const totalTrials = 10;
        let currentTrial = 0;
        let startTime = 0;
        let timerTimeout = null;
        let isWaitingForResponse = false;
        let currentType = null; // 'GO' veya 'NOGO'
        let trialData = [];

        // DOM Elemanları
        const targetCircle = document.getElementById('target-circle');
        const instruction = document.getElementById('instruction');
        const startBtn = document.getElementById('start-btn');
        const stopBtn = document.getElementById('stop-btn');
        const exportBtn = document.getElementById('export-btn');
        
        const statTrials = document.getElementById('stat-trials');
        const statLastRt = document.getElementById('stat-last-rt');
        const statAvgRt = document.getElementById('stat-avg-rt');
        const statErrors = document.getElementById('stat-errors');

        // Renk Kodları
        const COLOR_IDLE = '#333333';
        const COLOR_GO = '#00e676';     // Yeşil
        const COLOR_NOGO = '#ff1744';   // Kırmızı

        startBtn.addEventListener('click', startTest);
        stopBtn.addEventListener('click', stopTest);
        exportBtn.addEventListener('click', exportCSV);

        targetCircle.addEventListener('pointerdown', handleTap);

        function startTest() {
            if (!audioCtx) audioCtx = new AudioContext();
            trialData = [];
            currentTrial = 0;
            statErrors.textContent = '0';
            statLastRt.textContent = '0 ms';
            statAvgRt.textContent = '0 ms';
            
            startBtn.disabled = true;
            stopBtn.disabled = false;
            exportBtn.disabled = true;

            nextTrial();
        }

        function stopTest() {
            clearTimeout(timerTimeout);
            isWaitingForResponse = false;
            targetCircle.style.backgroundColor = COLOR_IDLE;
            targetCircle.textContent = 'DURDURULDU';
            instruction.textContent = 'Test durduruldu. Yeniden başlamak için butona basın.';
            
            startBtn.disabled = false;
            stopBtn.disabled = true;
            if (trialData.length > 0) exportBtn.disabled = false;
        }

        function nextTrial() {
            if (currentTrial >= totalTrials) {
                finishTest();
                return;
            }

            currentTrial++;
            statTrials.textContent = `${currentTrial}/${totalTrials}`;
            
            targetCircle.style.backgroundColor = COLOR_IDLE;
            targetCircle.textContent = 'BEKLEYİN...';
            instruction.textContent = 'Yeşil yanarsa DOKUNUN! Kırmızı yanarsa BEKLEYİN!';
            isWaitingForResponse = false;

            // Rastgele Bekleme Süresi (1.5 - 3.5 saniye arası)
            const randomDelay = Math.floor(Math.random() * 2000) + 1500;

            timerTimeout = setTimeout(() => {
                triggerStimulus();
            }, randomDelay);
        }

        function triggerStimulus() {
            // %70 GO (Yeşil), %30 NO-GO (Kırmızı) olasılığı
            const isGo = Math.random() < 0.7;
            currentType = isGo ? 'GO' : 'NOGO';

            if (isGo) {
                targetCircle.style.backgroundColor = COLOR_GO;
                targetCircle.textContent = 'DOKUN!';
                playGoSound();
            } else {
                targetCircle.style.backgroundColor = COLOR_NOGO;
                targetCircle.textContent = 'DUR!';
                playNoGoSound();
            }

            startTime = performance.now();
            isWaitingForResponse = true;

            // No-Go durumunda sporcu basmazsa 1.2 saniye sonra otomatik başarılı say ve geç
            if (!isGo) {
                timerTimeout = setTimeout(() => {
                    if (isWaitingForResponse && currentType === 'NOGO') {
                        // Başarılı Baskılama (Inhibition Success)
                        recordResult('NOGO', 0, true, 'Doğru Baskılama');
                        nextTrial();
                    }
                }, 1200);
            }
        }

        function handleTap(e) {
            e.preventDefault();
            if (!isWaitingForResponse) {
                // Erken dokunma (False Start)
                if (startBtn.disabled && targetCircle.textContent === 'BEKLEYİN...') {
                    clearTimeout(timerTimeout);
                    recordResult('EARLY', 0, false, 'Erken Dokunma');
                    targetCircle.textContent = 'ERKEN!';
                    playSound(100, 'sawtooth', 0.2);
                    setTimeout(nextTrial, 1000);
                }
                return;
            }

            const endTime = performance.now();
            const reactionTime = Math.round(endTime - startTime);
            isWaitingForResponse = false;
            clearTimeout(timerTimeout);

            if (currentType === 'GO') {
                // Doğru Reaksiyon
                recordResult('GO', reactionTime, true, 'Doğru Dokunma');
                statLastRt.textContent = reactionTime + ' ms';
                targetCircle.textContent = reactionTime + ' ms';
            } else if (currentType === 'NOGO') {
                // Hatalı Dokunma (Baskılama Hatası / Commission Error)
                recordResult('NOGO', reactionTime, false, 'Hatalı Dokunma (No-Go)');
                targetCircle.textContent = 'HATA!';
                playSound(100, 'sawtooth', 0.3);
            }

            updateStats();
            setTimeout(nextTrial, 1000);
        }

        function recordResult(type, rt, isCorrect, note) {
            trialData.push({
                trial: currentTrial,
                type: type,
                reactionTimeMs: rt,
                isCorrect: isCorrect,
                note: note
            });
        }

        function updateStats() {
            const validGoTrials = trialData.filter(d => d.type === 'GO' && d.isCorrect);
            const errors = trialData.filter(d => !d.isCorrect).length;

            statErrors.textContent = errors;

            if (validGoTrials.length > 0) {
                const avg = Math.round(validGoTrials.reduce((acc, curr) => acc + curr.reactionTimeMs, 0) / validGoTrials.length);
                statAvgRt.textContent = avg + ' ms';
            }
        }

        function finishTest() {
            targetCircle.style.backgroundColor = COLOR_IDLE;
            targetCircle.textContent = 'TAMAMLANDI';
            instruction.textContent = 'Test bitti! Verileri indirmek için "CSV İndir" butonuna basabilirsiniz.';
            
            startBtn.disabled = false;
            stopBtn.disabled = true;
            exportBtn.disabled = false;
        }

        function exportCSV() {
            if (trialData.length === 0) return;

            let csvContent = "data:text/csv;charset=utf-8,Deneme,Tip,Reaksiyon_Zamani_ms,Dogru_Mu,Aciklama\n";
            trialData.forEach(row => {
                csvContent += `${row.trial},${row.type},${row.reactionTimeMs},${row.isCorrect ? 1 : 0},${row.note}\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `SPPLab_Cognitive_Data_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }
    </script>
